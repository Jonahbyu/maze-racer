name: Build and deploy web build

on:
  push:
    branches: [main]
  workflow_dispatch:

# GITHUB_TOKEN needs these to publish to Pages.
permissions:
  contents: read
  pages: write
  id-token: write

# Never let two deploys race; let a newer push supersede an in-flight one.
concurrency:
  group: pages
  cancel-in-progress: true

env:
  GODOT_VERSION: 4.7

jobs:
  test:
    runs-on: ubuntu-latest
    container:
      # The image ships the engine and matching export templates already
      # installed, which is what makes the export step below a one-liner.
      image: barichello/godot-ci:4.7
    steps:
      - uses: actions/checkout@v4

      # A fresh clone has no .godot/ class cache, so every class_name fails to
      # resolve until the project is imported once. Same trap as launch.ps1.
      - name: Import project
        run: godot --headless --import || true

      - name: Rules harness
        run: |
          godot --headless --script res://scripts/core/RulesTest.gd 2>&1 | tee rules.log
          grep -qiE '(^| )FAIL|assertion failed' rules.log && exit 1 || true

      - name: Scene harness
        run: |
          godot --headless --script res://scripts/core/SceneTest.gd 2>&1 | tee scene.log
          grep -qiE '(^| )FAIL|assertion failed' scene.log && exit 1 || true

  build:
    needs: test
    runs-on: ubuntu-latest
    container:
      image: barichello/godot-ci:4.7
    steps:
      - uses: actions/checkout@v4

      - name: Install export templates
        run: |
          mkdir -p ~/.local/share/godot/export_templates
          mv /root/.local/share/godot/export_templates/* ~/.local/share/godot/export_templates/ 2>/dev/null || true
          ls ~/.local/share/godot/export_templates/ || true

      - name: Import project
        run: godot --headless --import || true

      - name: Export web build
        run: |
          mkdir -p build/web
          godot --headless --export-release "Web" build/web/index.html
          test -f build/web/index.wasm || { echo "export produced no wasm"; exit 1; }

      # A threaded build needs SharedArrayBuffer, which needs COOP/COEP headers,
      # which Pages cannot send. Fail loudly here rather than serving a blank canvas.
      - name: Assert nothreads build
        run: |
          if grep -q "pthread" build/web/index.js; then
            echo "ERROR: threaded build -- will not run on GitHub Pages"
            exit 1
          fi
          echo "nothreads build confirmed"

      # Pages skips files/dirs starting with an underscore without this.
      - name: Add .nojekyll
        run: touch build/web/.nojekyll

      - uses: actions/upload-pages-artifact@v3
        with:
          path: build/web

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
